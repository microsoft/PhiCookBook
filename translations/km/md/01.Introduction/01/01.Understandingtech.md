# បច្ចេកវិទ្យាសំខាន់ដែលបានចោលរួមមាន

1. [DirectML](https://learn.microsoft.com/windows/ai/directml/dml?WT.mc_id=aiml-138114-kinfeylo) - API កម្រិតទាបសម្រាប់ម៉ាស៊ីនរៀនជំរុញដោយឧបករណ៍រឹង ដែលបានបង្កើតនៅលើ DirectX 12។
2. [CUDA](https://blogs.nvidia.com/blog/what-is-cuda-2/) - វេទិកាការគណនាផ្សំគ្នា និងម៉ូដែលAPI ដែលបានបង្កើតដោយ Nvidia សម្រាប់អនុញ្ញាតឲ្យប្រើប្រាស់ការពារតំណពិនិត្យទូទៅលើឧបករណ៍ដំណើរការ ជាពិសេស GPUs។
3. [ONNX](https://onnx.ai/) (Open Neural Network Exchange) - រចនាប័ទ្មបើកសម្រាប់តំណាងគំរូម៉ាស៊ីនរៀន ដែលផ្តល់ការចងក្រងគ្នារវាងបន្ទុក ML ផ្សេងៗ។
4. [GGUF](https://github.com/ggerganov/ggml/blob/master/docs/gguf.md) (Generic Graph Update Format) - រចនាប័ទ្មដែលប្រើសម្រាប់តំណាង និងធ្វើបច្ចុប្បន្នភាពចំពោះគំរូម៉ាស៊ីនរៀន ជាពិសេសមានប្រយោជន៍សម្រាប់គំរូភាសាតូចៗដែលអាចដំណើរការបានល្អលើ CPU ជាមួយសំណុំបំលែងគុណភាព 4-8bit។

## DirectML

DirectML គឺជាអាង API កម្រិតទាប ដែលអនុញ្ញាតឲ្យម៉ាស៊ីនរៀនជំរុញដោយឧបករណ៍រឹង។ វាត្រូវបានសាងសង់លើ DirectX 12 ដើម្បីប្រើប្រាស់ការជំរុញ GPU ហើយគ្មានការទាក់ទាញទៅកាន់អ្នកផ្គត់ផ្គង់ណាមួយទេ ហើយមិនចាំបាច់ផ្លាស់ប្ដូរកូដសម្រាប់ធ្វើការជាមួយ GPU ផ្សេងៗ។ វាត្រូវបានប្រើសំខាន់សម្រាប់ការបណ្តុះបណ្តាលគំរូ និងការបង្ហាញលទ្ធផលលើ GPU។

សម្រាប់ការគាំទ្រឧបករណ៍រឹង DirectML ត្រូវបានរចនាឡើងសម្រាប់ធ្វើការជាមួយ GPU ជាច្រើនផ្ទុយគ្នា រួមមាន AMD GPU ចងក្រង និងបែកស្រទាប់, Intel GPU ចងក្រង និង Nvidia GPU បែកស្រទាប់។ វាជាផ្នែកមួយនៃវេទិកា AI Windows ហើយគាំទ្រក្នុង Windows 10 និង 11 អាចប្រើសម្រាប់បណ្តុះបណ្តាលគំរូនិងបង្ហាញលទ្ធផលលើឧបករណ៍ Windows អ្វីក៏បាន។

មានការកែលម្អ និងឱកាសជាច្រើនទាក់ទងនឹង DirectML ដូចជា ការគាំទ្រ អភិបាល ONNX ដល់ 150 ប្រភេទ និងត្រូវបានប្រើដោយ runtime ONNX និង WinML។ វាទទួលការគាំទ្រ​ពី​អ្នកផ្គត់ផ្គង់ហ្វីល ឧបករណ៍រឹងធំៗជាច្រើន ដែលនីមួយៗអនុវត្តន៍អំពើបញ្ជា metacommands ផ្សេងៗ។

## CUDA

CUDA ដែលមានន័យថា Compute Unified Device Architecture គឺជាវេទិកាសម្រាប់ការគណនាផ្សំគ្នា និងម៉ូដែល API ដែលបានបង្កើតដោយ Nvidia។ វាអនុញ្ញាតឲ្យអ្នកអភិវឌ្ឍន៍កម្មវិធីប្រើប្រាស់ GPU ដែលគាំទ្រ CUDA សម្រាប់ការចាត់ចែងទូទៅ - វិធីសាស្រ្តនេះហៅថា GPGPU (General-Purpose computing on Graphics Processing Units)। CUDA ជាឧបករណ៍សំខាន់សម្រាប់ជំរុញ GPU របស់ Nvidia ហើយត្រូវបានប្រើយ៉ាងទូលំទូលាយក្នុងវិស័យជាច្រើន រួមមានម៉ាស៊ីនរៀន ការគណនាវិទ្យាសាស្ត្រ និង​ការប្រតិបត្តិចល័តវីដេអូ។

ក្ដារត្រួតពិនិត្យរឹងសម្រាប់ CUDA គឺជាការគាំទ្រ GPU របស់ Nvidia ដោយផ្ទាល់ ដោយវាជាបច្ចេកទេសម្ចាស់កម្មសិទ្ធិ របស់ Nvidia។ រៀបចំត្រឹមត្រូវនិងស្ថាបត្យកម្មនីមួយៗគាំទ្រកម្មវិធី CUDA toolkit ជាក់លាក់ ដែលផ្តល់បណ្ណាល័យ និងឧបករណ៍ដែលចាំបាច់សម្រាប់អ្នកអភិវឌ្ឍន៍ក្នុងការបង្កើត និងដំណើរការកម្មវិធី CUDA។

## ONNX

ONNX (Open Neural Network Exchange) គឺជារចនាប័ទ្មបើកសម្រាប់តំណាងគំរូម៉ាស៊ីនរៀន។ វាបង្កើតនូវនិយមន័យរបស់គំរូបណ្តាញគណនាដែលអាចពង្រីកបាន ព្រមទាំងនិយមន័យប្រតិបត្តិការ built-in និងប្រភេទទិន្នន័យស្ដង់ដារ។ ONNX អនុញ្ញាតឲ្យអ្នកអភិវឌ្ឍន៍ផ្លាស់ប្តូរក model រវាងបន្ទុក ML ផ្សេងៗ ដើម្បីអាចធ្វើបច្ចេកទេសអន្តរកម្ម និងងាយស្រួលក្នុងការបង្កើត និងដាក់ឲ្យដំណើរការ អ្នកប្រើប្រាស់ AI។

Phi3 mini អាចដំណើរការជាមួយ ONNX Runtime លើ CPU និង GPU នៅលើឧបករណ៍ផ្សេងៗ រួមមានវេទិកាសើវ៉ែ រួមទាំង Windows, Linux និង Mac ដេស្ដុប និង CPU មេប៉ាល់។
ការកំណត់រចនាសម្ព័ន្ធដែលបានបន្ថែមមាន

- គំរូ ONNX សម្រាប់ int4 DML: ប្រើបច្ចេកទេស Quantized ទៅ int4 តាមរយៈ AWQ
- គំរូ ONNX សម្រាប់ fp16 CUDA
- គំរូ ONNX សម្រាប់ int4 CUDA: ប្រើបច្ចេកទេស Quantized ទៅ int4 តាមរយៈ RTN
- គំរូ ONNX សម្រាប់ int4 CPU និង Mobile: ប្រើបច្ចេកទេស Quantized ទៅ int4 តាមរយៈ RTN

## Llama.cpp

Llama.cpp គឺជាបណ្ណាល័យកម្មវិធីបើកប្រភព ដែលសរសេរជាភាសា C++។ វាធ្វើការបញ្ចេញលទ្ធផលលើគំរូភាសាធំៗ (LLMs) ជាច្រើន រួមមាន Llama។ បានអភិវឌ្ឍន៍ជាមួយបណ្ណាល័យ ggml (បណ្ណាល័យ General-purpose tensor) llama.cpp មានគោលបំណងផ្តល់នូវការបញ្ចេញលទ្ធផលរហ័ស និងការប្រើប្រាស់ម៉េមរីទាបជាងការអនុវត្ត Python ដើម។ វាគាំទ្រការជំរុញរឹង ការប្រែប្រួលគុណភាព និងផ្តល់ API និងឧទាហរណ៍ងាយស្រួល។ ប្រសិនបើអ្នកចាប់អារម្មណ៍លើការបញ្ចេញលទ្ធផល LLM ប្រសិទ្ធ ព្យមានថា llama.cpp គួរតែត្រូវបានស្វែងយល់ព្រោះ Phi3 អាចដំណើរការ Llama.cpp បាន។

## GGUF

GGUF (Generic Graph Update Format) គឺជារចនាប័ទ្មដែលប្រើសម្រាប់តំណាង និងធ្វើបច្ចុប្បន្នភាពលើគំរូម៉ាស៊ីនរៀន។ វាមានប្រយោជន៍ជាចម្បងចំពោះគំរូភាសាតូចៗ (SLMs) ដែលអាចដំណើរការបានប្រសើរលើ CPU ជាមួយការប្រែប្រួលគុណភាព 4-8bit។ GGUF មានអត្ថប្រយោជន៍សម្រាប់ការបង្កើតគំរូដំបូង​រហ័ស និងការរត់គំរូលើឧបករណ៍ខាងចុង ឬក៏ក្នុងការងារប្រមូលផ្តុំនានា ដូចជា CI/CD pipelines។
